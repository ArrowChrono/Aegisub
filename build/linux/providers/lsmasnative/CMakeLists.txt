# Local Linux/Clang19 port of the upstream build.zig source/link graph.
cmake_minimum_required(VERSION 3.20)
project(lsmasnative_linux C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This local recipe is for Linux only")
endif()
set(LSMAS_SOURCE "" CACHE PATH "Pinned upstream LsmasSharp project")
set(LSW_SOURCE "" CACHE PATH "Upstream prepared HOE sources")
set(FFMPEG_PREFIX "" CACHE PATH "Patched FFmpeg static prefix")
set(LSW_COMMON utils osdep decode lwlibav_dec lwlibav_video lwlibav_audio lwindex video_output audio_output resample lwindex_parser lwindex_utils planar_yuv_sse2)
set(SOURCES "${LSMAS_SOURCE}/src/LsmasNative/lsmas_native.c" "${LSW_SOURCE}/xxHash/xxhash.c")
foreach(name IN LISTS LSW_COMMON)
  list(APPEND SOURCES "${LSW_SOURCE}/common/${name}.c")
endforeach()
if(EXISTS "${LSW_SOURCE}/common/qsv.c")
  list(APPEND SOURCES "${LSW_SOURCE}/common/qsv.c")
endif()
add_library(lsmasnative SHARED ${SOURCES})
set_target_properties(lsmasnative PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(lsmasnative PRIVATE "${LSW_SOURCE}/common" "${LSW_SOURCE}/xxHash" "${FFMPEG_PREFIX}/include")
target_compile_features(lsmasnative PRIVATE c_std_11)
target_compile_options(lsmasnative PRIVATE -Wno-deprecated-declarations)
target_compile_definitions(lsmasnative PRIVATE _FILE_OFFSET_BITS=64 XXH_INLINE_ALL LSMAS_LSW_VARIANT_HOE=1
 LSMAS_NATIVE_VERSION="202608+linux-clang19" LSMAS_NATIVE_GIT_HEAD="4c85eb4c68e3" LSMAS_NATIVE_ZIG_VERSION="not-used"
 LSMASHWORKS_GIT_URL="https://github.com/HomeOfAviSynthPlusEvolution/L-SMASH-Works.git" LSMASHWORKS_GIT_HEAD="7e65185d3f08" LSMASHWORKS_GIT_BRANCH="1310.0.0.0"
 FFMPEG_GIT_URL="https://github.com/HomeOfAviSynthPlusEvolution/FFmpeg" FFMPEG_GIT_HEAD="39c24f36247f+mov-audio-end" FFMPEG_GIT_BRANCH="custom-patches-for-lsmashsource"
 DAV1D_VERSION="1.5.3" ZLIB_VERSION_STR="1.3.1")
set(FFMPEG_ARCHIVES)
foreach(name avformat avcodec swscale swresample avutil dav1d z)
  list(APPEND FFMPEG_ARCHIVES "${FFMPEG_PREFIX}/lib/lib${name}.a")
endforeach()
target_link_libraries(lsmasnative PRIVATE "-Wl,--start-group" ${FFMPEG_ARCHIVES} "-Wl,--end-group" pthread dl m atomic)
target_link_options(lsmasnative PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,defs" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
install(TARGETS lsmasnative LIBRARY DESTINATION lib)
install(FILES "${LSMAS_SOURCE}/src/LsmasNative/lsmas_native.h" DESTINATION include)
