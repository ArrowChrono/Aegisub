    # Keep the existing custom targets as developer entry points, but let CTest
    # execute the smoke command itself so timeout and process cleanup are owned
    # by the test runner rather than a nested build.
    # Script-based tests intentionally follow the repository's build-then-test
    # contract; CTest does not infer custom managed-output dependencies from
    # paths passed as script arguments.
    function(aegisub_set_smoke_test_properties test_name timeout_seconds labels)
        set_tests_properties("${test_name}" PROPERTIES
            LABELS "${labels}"
            RUN_SERIAL TRUE
            TIMEOUT "${timeout_seconds}"
        )
    endfunction()

    add_test(
