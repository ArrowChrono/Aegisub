    # Sources which still belong to the application target should use the
    # exact production compile environment, while being compiled only once
    # when CI builds Aegisub before gtest-run.
    get_target_property(_aegisub_executable_sources Aegisub SOURCES)
    get_target_property(_gtest_target_sources gtest-run SOURCES)
    set(_aegisub_test_production_sources)
    set(_aegisub_test_production_source_keys)
    foreach(_test_source IN LISTS _gtest_target_sources)
        if(_test_source MATCHES "^\\$<"
            OR NOT _test_source MATCHES "\\.(c|cc|cpp|cxx|m|mm)$")
            continue()
        endif()

        get_filename_component(_test_source_absolute "${_test_source}"
            ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        set(_test_source_key "${_test_source_absolute}")
        if(WIN32)
            string(TOLOWER "${_test_source_key}" _test_source_key)
        endif()

        foreach(_executable_source IN LISTS _aegisub_executable_sources)
            if(_executable_source MATCHES "^\\$<")
                continue()
            endif()
            get_filename_component(_executable_source_absolute
                "${_executable_source}" ABSOLUTE
                BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
            set(_executable_source_key "${_executable_source_absolute}")
            if(WIN32)
                string(TOLOWER "${_executable_source_key}"
                    _executable_source_key)
            endif()
            if(_test_source_key STREQUAL _executable_source_key)
                list(FIND _aegisub_test_production_source_keys
                    "${_test_source_key}" _shared_source_index)
                if(_shared_source_index EQUAL -1)
                    list(APPEND _aegisub_test_production_sources
                        "${_test_source_absolute}")
                    list(APPEND _aegisub_test_production_source_keys
                        "${_test_source_key}")
                endif()
                break()
            endif()
        endforeach()
    endforeach()

    if(_aegisub_test_production_sources)
        set(_aegisub_executable_sources_filtered)
        foreach(_source IN LISTS _aegisub_executable_sources)
            if(_source MATCHES "^\\$<")
                list(APPEND _aegisub_executable_sources_filtered "${_source}")
                continue()
            endif()
            get_filename_component(_source_absolute "${_source}"
                ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
            set(_source_key "${_source_absolute}")
            if(WIN32)
                string(TOLOWER "${_source_key}" _source_key)
            endif()
            list(FIND _aegisub_test_production_source_keys
                "${_source_key}" _shared_source_index)
            if(_shared_source_index EQUAL -1)
                list(APPEND _aegisub_executable_sources_filtered "${_source}")
            endif()
        endforeach()

        set(_gtest_target_sources_filtered)
        foreach(_source IN LISTS _gtest_target_sources)
            if(_source MATCHES "^\\$<")
                list(APPEND _gtest_target_sources_filtered "${_source}")
                continue()
            endif()
            get_filename_component(_source_absolute "${_source}"
                ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
            set(_source_key "${_source_absolute}")
            if(WIN32)
                string(TOLOWER "${_source_key}" _source_key)
            endif()
            list(FIND _aegisub_test_production_source_keys
                "${_source_key}" _shared_source_index)
            if(_shared_source_index EQUAL -1)
                list(APPEND _gtest_target_sources_filtered "${_source}")
            endif()
        endforeach()

                set_property(TARGET aegisub_test_production_objects
                    PROPERTY ${_property} "${_property_value}")
            endif()
        endforeach()

        get_target_property(_aegisub_link_libraries Aegisub LINK_LIBRARIES)
        if(NOT "${_aegisub_link_libraries}" STREQUAL
            "_aegisub_link_libraries-NOTFOUND")
            target_link_libraries(aegisub_test_production_objects PRIVATE
                ${_aegisub_link_libraries})
        endif()
        get_target_property(_aegisub_dependencies Aegisub
            MANUALLY_ADDED_DEPENDENCIES)
        if(NOT "${_aegisub_dependencies}" STREQUAL
            "_aegisub_dependencies-NOTFOUND")
            add_dependencies(aegisub_test_production_objects
                ${_aegisub_dependencies})
        endif()

        get_target_property(_aegisub_precompile_headers Aegisub
            PRECOMPILE_HEADERS)
        if(NOT "${_aegisub_precompile_headers}" STREQUAL
            "_aegisub_precompile_headers-NOTFOUND"
            AND NOT "${_aegisub_precompile_headers}" STREQUAL "")
            target_precompile_headers(aegisub_test_production_objects PRIVATE
                ${_aegisub_precompile_headers})
        endif()

        target_link_libraries(Aegisub PRIVATE
            aegisub_test_production_objects)
        target_link_libraries(gtest-run PRIVATE
            aegisub_test_production_objects)

        list(LENGTH _aegisub_test_production_sources
            _aegisub_test_production_source_count)
        message(STATUS
            "gtest-run reuses ${_aegisub_test_production_source_count} Aegisub production objects")
    endif()
    target_link_libraries(gtest-run PRIVATE aegisub_core aegisub_features libaegisub luajit CLI11::CLI11 "GTest::GTest" "Iconv::Iconv" pugixml::pugixml "ZLIB::ZLIB" "Boost::locale" ${wxWidgets_LIBRARIES})
    target_include_directories(gtest-run PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/src/libresrc")
    target_include_directories(gtest-run PRIVATE "${PROJECT_SOURCE_DIR}/vendor")
    aegisub_enable_highway(gtest-run)
    if(WIN32)
        target_link_libraries(gtest-run PRIVATE Psapi)
    endif()
    target_precompile_headers(gtest-run PRIVATE "${PROJECT_SOURCE_DIR}/tests/support/tests_pre.h")
    # Generated fixtures and gtest CWD live under the build tree, not the source tree.
    set(AEGISUB_TEST_RUNTIME_DIR "${CMAKE_CURRENT_BINARY_DIR}/tests-runtime")
    file(MAKE_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}")
    if(WIN32)
        target_link_libraries(gtest-run PRIVATE ws2_32)
        add_custom_target(test-aegisub
            COMMAND "${PROJECT_SOURCE_DIR}/tests/setup.bat"
            COMMAND gtest-run
            WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
        )
    else()
        add_custom_target(test-aegisub
            COMMAND "${PROJECT_SOURCE_DIR}/tests/setup.sh"
            COMMAND gtest-run
            WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
        )
    endif()
    if(BUILD_TESTING)
        if(WIN32)
            add_test(
                NAME aegisub_test_data_setup
                COMMAND "${PROJECT_SOURCE_DIR}/tests/setup.bat"
                WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
            )
        else()
            add_test(
                NAME aegisub_test_data_setup
                COMMAND "${PROJECT_SOURCE_DIR}/tests/setup.sh"
                WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
            )
        endif()
        set_tests_properties(aegisub_test_data_setup PROPERTIES
            FIXTURES_SETUP aegisub_test_data
            LABELS "setup"
        )

        include(GoogleTest)
        gtest_discover_tests(gtest-run
            WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
            DISCOVERY_TIMEOUT 60
            PROPERTIES
                FIXTURES_REQUIRED aegisub_test_data
                LABELS "gtest"
        )
    endif()

    add_executable(font-collector-state-test EXCLUDE_FROM_ALL
        tests/tests/font_collector_state.cpp
    )
    target_compile_features(font-collector-state-test PRIVATE cxx_std_20)

        aegisub_core

    )
    if(BUILD_TESTING)
        add_test(
            NAME font_collector_state
            COMMAND "$<TARGET_FILE:font-collector-state-test>"
            WORKING_DIRECTORY "${AEGISUB_TEST_RUNTIME_DIR}"
        )
        set_tests_properties(font_collector_state PROPERTIES LABELS "gtest")
    endif()

    add_custom_target(test-aegisub)
